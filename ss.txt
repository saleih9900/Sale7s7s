-- ============================================
--  جداول مكتب باداؤود — نسخة نظيفة
-- ============================================

-- 1) الإعدادات
create table settings (
  id int primary key default 1,
  "officeName" text default 'مكتب باداؤود للهندسة والمقاولات',
  tagline text default 'نصمم فضاءً يُلهم الحياة',
  phone text default '',
  email text default '',
  address text default '',
  about text default '',
  updated_at timestamptz default now(),
  constraint single_row check (id = 1)
);

insert into settings (id) values (1) on conflict (id) do nothing;

-- 2) إعدادات الواجهة
create table ui_settings (
  id int primary key default 1,
  site_name text default 'مكتب باداؤود للهندسة والمقاولات',
  logo_url text default '',
  primary_color text default '#6366f1',
  default_theme text default 'dark',
  show_services boolean default true,
  show_projects boolean default true,
  constraint ui_single_row check (id = 1)
);

insert into ui_settings (id) values (1) on conflict (id) do nothing;

-- 3) الخدمات
create table services (
  id bigserial primary key,
  icon text default '✦',
  title text not null,
  description text not null,
  created_at timestamptz default now()
);

-- 4) المشاريع
create table projects (
  id bigserial primary key,
  slug text unique,
  icon text default '⌂',
  cat text default 'سكني',
  title text not null,
  description text not null,
  content text default '',
  cover_image text default '',
  gallery text[] default '{}',
  location text default '',
  year text default '',
  area text default '',
  client text default '',
  created_at timestamptz default now()
);

-- 5) الرسائل
create table messages (
  id bigserial primary key,
  name text,
  email text,
  phone text,
  message text,
  created_at timestamptz default now()
);

-- 6) بيانات تجريبية
insert into services (icon, title, description) values
  ('✎', 'التصميم المعماري', 'تصاميم عصرية مبتكرة'),
  ('⌂', 'الإشراف الهندسي', 'متابعة دقيقة للتنفيذ'),
  ('◈', 'التصميم الداخلي', 'حلول داخلية أنيقة');

-- 7) تفعيل RLS
alter table settings enable row level security;
alter table ui_settings enable row level security;
alter table services enable row level security;
alter table projects enable row level security;
alter table messages enable row level security;

-- 8) سياسات settings
create policy "settings_read" on settings for select using (true);
create policy "settings_write" on settings for all to authenticated using (true) with check (true);

-- 9) سياسات ui_settings
create policy "ui_read" on ui_settings for select using (true);
create policy "ui_write" on ui_settings for all to authenticated using (true) with check (true);

-- 10) سياسات services
create policy "services_read" on services for select using (true);
create policy "services_write" on services for all to authenticated using (true) with check (true);

-- 11) سياسات projects
create policy "projects_read" on projects for select using (true);
create policy "projects_write" on projects for all to authenticated using (true) with check (true);

-- 12) سياسات messages
create policy "messages_insert" on messages for insert to anon, authenticated with check (true);
create policy "messages_read" on messages for select to authenticated using (true);
create policy "messages_delete" on messages for delete to authenticated using (true);
